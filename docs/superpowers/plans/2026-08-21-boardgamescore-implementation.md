# Board Game Score Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build Board Game Score — a client-side-only React app for tracking board game scores via user-definable scoring templates (categories + optional formula + win direction), a persistent player roster, and browsable game history.

**Architecture:** A framework-free, unit-tested pure-logic layer under `src/logic/` (formula parsing/eval, template validation/standings, game state transitions, persistence/backup), consumed by a single `useAppState` hook (`useReducer` + localStorage sync), rendered by MUI-based React components under `src/components/`, one subfolder per screen.

**Tech Stack:** React 19 + Vite + TypeScript + MUI, Vitest + Testing Library (unit/component tests), Playwright (e2e), oxlint + prettier, GitHub Actions → GitHub Pages.

**Spec:** `docs/superpowers/specs/2026-08-21-boardgamescore-design.md`

## Global Constraints

- No `eval`/`Function` anywhere — formulas are parsed and evaluated by the hand-written parser in `src/logic/formula.ts`.
- All logic-layer functions (`src/logic/**`) are pure: no `Date.now()`, `crypto.randomUUID()`, or `localStorage` access inside them. IDs and timestamps are always generated by the caller (component or hook) and passed in as parameters.
- `formatVersion: 1` in the backup format; `.prettierrc` is `{ "singleQuote": true, "semi": true, "tabWidth": 2, "printWidth": 100 }`; base path is `/boardgamescore/`.
- TDD for every file under `src/logic/`: write the failing test before the implementation.
- Every screen component ends up under `src/components/<ScreenName>/`.

```ts
export type FormulaNode =
  | { type: 'num'; value: number }
  | { type: 'ident'; key: string }
  | { type: 'neg'; operand: FormulaNode }
  | { type: 'binary'; op: '+' | '-' | '*' | '/'; left: FormulaNode; right: FormulaNode };

export type ParseResult = { ok: true; ast: FormulaNode } | { ok: false; error: string };

type Token =
  | { type: 'num'; value: number }
  | { type: 'ident'; value: string }
  | { type: 'op'; value: '+' | '-' | '*' | '/' }
  | { type: 'lparen' }
  | { type: 'rparen' };

function tokenize(source: string): Token[] | { error: string } {
  const tokens: Token[] = [];
  let i = 0;
  while (i < source.length) {
    const ch = source[i];
    if (/\s/.test(ch)) {
      i += 1;
      continue;
    }
    if (/[0-9.]/.test(ch)) {
      let j = i;
      while (j < source.length && /[0-9.]/.test(source[j])) j += 1;
      const value = Number(source.slice(i, j));
      if (Number.isNaN(value)) return { error: `Invalid number near position ${i}` };
      tokens.push({ type: 'num', value });
      i = j;
      continue;
    }
    if (/[A-Za-z_]/.test(ch)) {
      let j = i;
      while (j < source.length && /[A-Za-z0-9_]/.test(source[j])) j += 1;
      tokens.push({ type: 'ident', value: source.slice(i, j) });
      i = j;
      continue;
    }
    if (ch === '+' || ch === '-' || ch === '*' || ch === '/') {
      tokens.push({ type: 'op', value: ch });
      i += 1;
      continue;
    }
    if (ch === '(') {
      tokens.push({ type: 'lparen' });
      i += 1;
      continue;
    }
    if (ch === ')') {
      tokens.push({ type: 'rparen' });
      i += 1;
      continue;
    }
    return { error: `Unexpected character "${ch}" near position ${i}` };
  }
  return tokens;
}

class Parser {
  private pos = 0;
  constructor(
    private tokens: Token[],
    private knownKeys: Set<string>,
  ) {}

  private peek(): Token | undefined {
    return this.tokens[this.pos];
  }

  private next(): Token | undefined {
    return this.tokens[this.pos++];
  }

  parseExpr(): FormulaNode {
    let node = this.parseTerm();
    for (;;) {
      const tok = this.peek();
      if (tok?.type === 'op' && (tok.value === '+' || tok.value === '-')) {
        this.next();
        const right = this.parseTerm();
        node = { type: 'binary', op: tok.value, left: node, right };
      } else {
        break;
      }
    }
    return node;
  }

  private parseTerm(): FormulaNode {
    let node = this.parseFactor();
    for (;;) {
      const tok = this.peek();
      if (tok?.type === 'op' && (tok.value === '*' || tok.value === '/')) {
        this.next();
        const right = this.parseFactor();
        node = { type: 'binary', op: tok.value, left: node, right };
      } else {
        break;
      }
    }
    return node;
  }

  private parseFactor(): FormulaNode {
    const tok = this.next();
    if (!tok) throw new Error('Unexpected end of formula');
    if (tok.type === 'op' && tok.value === '-') {
      return { type: 'neg', operand: this.parseFactor() };
    }
    if (tok.type === 'num') {
      return { type: 'num', value: tok.value };
    }
    if (tok.type === 'ident') {
      if (!this.knownKeys.has(tok.value)) {
        throw new Error(`Unknown category "${tok.value}" in formula`);
      }
      return { type: 'ident', key: tok.value };
    }
    if (tok.type === 'lparen') {
      const inner = this.parseExpr();
      const close = this.next();
      if (close?.type !== 'rparen') throw new Error("Expected ')'");
      return inner;
    }
    throw new Error('Unexpected token in formula');
  }

  parseAndCheckComplete(): FormulaNode {
    const node = this.parseExpr();
    if (this.pos < this.tokens.length) {
      throw new Error('Unexpected trailing tokens in formula');
    }
    return node;
  }
}

export function parseFormula(source: string, knownKeys: string[]): ParseResult {
  if (!source.trim()) return { ok: false, error: 'Formula is empty' };
  const tokens = tokenize(source);
  if (!Array.isArray(tokens)) return { ok: false, error: tokens.error };
  if (tokens.length === 0) return { ok: false, error: 'Formula is empty' };
  try {
    const parser = new Parser(tokens, new Set(knownKeys));
    const ast = parser.parseAndCheckComplete();
    return { ok: true, ast };
  } catch (err) {
    return { ok: false, error: err instanceof Error ? err.message : 'Invalid formula' };
  }
}

export function evaluateFormula(ast: FormulaNode, values: Record<string, number>): number {
  switch (ast.type) {
    case 'num':
      return ast.value;
    case 'ident':
      return values[ast.key] ?? 0;
    case 'neg':
      return -evaluateFormula(ast.operand, values);
    case 'binary': {
      const left = evaluateFormula(ast.left, values);
      const right = evaluateFormula(ast.right, values);
      switch (ast.op) {
        case '+':
          return left + right;
        case '-':
          return left - right;
        case '*':
          return left * right;
        case '/':
          return right === 0 ? 0 : left / right;
      }
    }
  }
}
```

```ts
import { evaluateFormula, parseFormula } from './formula';
import { PRESET_TEMPLATES } from './presetTemplates';
import type { Category, GameSession, Player, ScoreEntry, StandingRow, Template } from './types';

export function slugifyKey(name: string): string {
  const words = name
    .trim()
    .split(/[^a-zA-Z0-9]+/)
    .filter(Boolean);
  if (words.length === 0) return '';
  return words
    .map((word, index) => {
      const lower = word.toLowerCase();
      return index === 0 ? lower : lower.charAt(0).toUpperCase() + lower.slice(1);
    })
    .join('');
}

export function categorySubtotal(entries: ScoreEntry[] | undefined): number {
  if (!entries) return 0;
  return entries.reduce((sum, entry) => sum + entry.value, 0);
}

export function computeCategoryTotalsById(
  template: Template,
  game: GameSession,
  playerId: string,
): Record<string, number> {
  const result: Record<string, number> = {};
  for (const category of template.categories) {
    result[category.id] = categorySubtotal(game.entries[playerId]?.[category.id]);
  }
  return result;
}

export function computePlayerTotal(
  template: Template,
  categoryTotalsById: Record<string, number>,
): number {
  if (!template.formula) {
    return Object.values(categoryTotalsById).reduce((sum, value) => sum + value, 0);
  }
  const keys = template.categories.map((c) => c.key);
  const parsed = parseFormula(template.formula, keys);
  if (!parsed.ok) return 0;
  const values: Record<string, number> = {};
  for (const category of template.categories) {
    values[category.key] = categoryTotalsById[category.id] ?? 0;
  }
  return evaluateFormula(parsed.ast, values);
}

export function computeStandings(
  template: Template,
  game: GameSession,
  players: Player[],
): StandingRow[] {
  const rows = game.playerIds.map((playerId) => {
    const player = players.find((p) => p.id === playerId);
    const categoryTotalsById = computeCategoryTotalsById(template, game, playerId);
    const categoryTotals = template.categories.map((category) => ({
      categoryId: category.id,
      categoryName: category.name,
      total: categoryTotalsById[category.id],
    }));
    return {
      playerId,
      playerName: player?.name ?? 'Unknown Player',
      categoryTotals,
      total: computePlayerTotal(template, categoryTotalsById),
      rank: 0,
    };
  });

  const sorted = rows.slice().sort((a, b) => {
    return template.winDirection === 'high' ? b.total - a.total : a.total - b.total;
  });

  let rank = 0;
  let prevTotal: number | null = null;
  return sorted.map((row, index) => {
    if (prevTotal === null || row.total !== prevTotal) {
      rank = index + 1;
      prevTotal = row.total;
    }
    return { ...row, rank };
  });
}

export interface ValidateTemplateInput {
  name: string;
  categories: Category[];
  formula: string | null;
}

export type ValidateTemplateResult = { ok: true } | { ok: false; error: string };

export function validateTemplate(input: ValidateTemplateInput): ValidateTemplateResult {
  if (!input.name.trim()) return { ok: false, error: 'Template name is required.' };
  if (input.categories.length === 0) {
    return { ok: false, error: 'At least one category is required.' };
  }
  const keys = new Set<string>();
  for (const category of input.categories) {
    if (!category.name.trim()) return { ok: false, error: 'Every category needs a name.' };
    if (!category.key) return { ok: false, error: `Category "${category.name}" has no valid key.` };
    if (keys.has(category.key)) {
      return { ok: false, error: `Duplicate category key "${category.key}".` };
    }
    keys.add(category.key);
  }
  if (input.formula) {
    const parsed = parseFormula(
      input.formula,
      input.categories.map((c) => c.key),
    );
    if (!parsed.ok) return { ok: false, error: parsed.error };
  }
  return { ok: true };
}

export function getAllTemplates(customTemplates: Template[]): Template[] {
  return [...PRESET_TEMPLATES, ...customTemplates];
}
```

```ts
import type { GameSession } from './types';

export function createGame(
  id: string,
  templateId: string,
  playerIds: string[],
  nowIso: string,
): GameSession {
  return {
    id,
    templateId,
    playerIds,
    entries: {},
    moveLog: [],
    startedAt: nowIso,
    finishedAt: null,
  };
}

export function addEntry(
  game: GameSession,
  playerId: string,
  categoryId: string,
  value: number,
  entryId: string,
): GameSession {
  const playerEntries = game.entries[playerId] ?? {};
  const categoryEntries = playerEntries[categoryId] ?? [];
  return {
    ...game,
    entries: {
      ...game.entries,
      [playerId]: {
        ...playerEntries,
        [categoryId]: [...categoryEntries, { id: entryId, value }],
      },
    },
    moveLog: [...game.moveLog, { playerId, categoryId, entryId }],
  };
}

export function removeEntry(
  game: GameSession,
  playerId: string,
  categoryId: string,
  entryId: string,
): GameSession {
  const playerEntries = game.entries[playerId];
  if (!playerEntries) return game;
  const categoryEntries = playerEntries[categoryId];
  if (!categoryEntries) return game;
  return {
    ...game,
    entries: {
      ...game.entries,
      [playerId]: {
        ...playerEntries,
        [categoryId]: categoryEntries.filter((entry) => entry.id !== entryId),
      },
    },
    moveLog: game.moveLog.filter((move) => move.entryId !== entryId),
  };
}

export function undoLastEntry(game: GameSession): GameSession {
  const last = game.moveLog[game.moveLog.length - 1];
  if (!last) return game;
  return removeEntry(game, last.playerId, last.categoryId, last.entryId);
}

export function finishGame(game: GameSession, nowIso: string): GameSession {
  return { ...game, finishedAt: nowIso };
}
```

```ts
import type { GameSession, HistoryEntry, Player, Template } from './types';

export interface AppState {
  players: Player[];
  templates: Template[];
  game: GameSession | null;
  history: HistoryEntry[];
}

export interface Backup extends AppState {
  app: 'boardgamescore';
  formatVersion: 1;
  exportedAt: string;
}

export function serialize(value: unknown): string {
  return JSON.stringify(value);
}

export function deserialize<T>(str: string | null): T | null {
  try {
    return str == null ? null : (JSON.parse(str) as T);
  } catch {
    return null;
  }
}

export function isResumable(game: GameSession | null): boolean {
  return !!game && game.finishedAt === null;
}

export function buildBackup(state: AppState, nowMs: number): Backup {
  return {
    app: 'boardgamescore',
    formatVersion: 1,
    exportedAt: new Date(nowMs).toISOString(),
    players: state.players,
    templates: state.templates,
    game: state.game,
    history: state.history,
  };
}

export type ValidateBackupResult =
  | { ok: true; backup: AppState }
  | { ok: false; reason: string };

export function validateBackup(obj: unknown): ValidateBackupResult {
  const notBackup: ValidateBackupResult = {
    ok: false,
    reason: 'Not a Board Game Score backup file.',
  };
  if (!obj || typeof obj !== 'object') return notBackup;
  const o = obj as Record<string, unknown>;
  if (o.app !== 'boardgamescore') return notBackup;
  if (typeof o.formatVersion === 'number' && o.formatVersion > 1) {
    return { ok: false, reason: 'This backup was made by a newer version of Board Game Score.' };
  }
  const players = Array.isArray(o.players) ? (o.players as Player[]) : [];
  const templates = Array.isArray(o.templates) ? (o.templates as Template[]) : [];
  const history = Array.isArray(o.history) ? (o.history as HistoryEntry[]) : [];
  return {
    ok: true,
    backup: {
      players: players.filter((p) => p && typeof p === 'object'),
      templates: templates.filter((t) => t && typeof t === 'object'),
      history: history.filter((h) => h && typeof h === 'object'),
      game: o.game && typeof o.game === 'object' ? (o.game as GameSession) : null,
    },
  };
}

function upsertById<T extends { id: string }>(list: T[], item: T): T[] {
  const arr = list.slice();
  const i = arr.findIndex((x) => x.id === item.id);
  if (i >= 0) arr[i] = item;
  else arr.push(item);
  return arr;
}

function upsertHistory(history: HistoryEntry[], entry: HistoryEntry): HistoryEntry[] {
  const arr = history.slice();
  const i = arr.findIndex((h) => h.gameId === entry.gameId);
  if (i >= 0) arr[i] = entry;
  else arr.push(entry);
  return arr;
}

export interface MergeSummary {
  playersAdded: number;
  playersUpdated: number;
  templatesAdded: number;
  templatesUpdated: number;
  gamesAdded: number;
  gamesUpdated: number;
  gameRestored: boolean;
  gameSkipped: boolean;
}

export function mergeBackup(
  state: AppState,
  backup: AppState,
): { state: AppState; summary: MergeSummary } {
  let players = state.players;
  let playersAdded = 0;
  let playersUpdated = 0;
  for (const p of backup.players) {
    if (players.some((x) => x.id === p.id)) playersUpdated += 1;
    else playersAdded += 1;
    players = upsertById(players, p);
  }

  let templates = state.templates;
  let templatesAdded = 0;
  let templatesUpdated = 0;
  for (const t of backup.templates) {
    if (templates.some((x) => x.id === t.id)) templatesUpdated += 1;
    else templatesAdded += 1;
    templates = upsertById(templates, t);
  }

  let history = state.history;
  let gamesAdded = 0;
  let gamesUpdated = 0;
  for (const h of backup.history) {
    if (history.some((x) => x.gameId === h.gameId)) gamesUpdated += 1;
    else gamesAdded += 1;
    history = upsertHistory(history, h);
  }

  let game = state.game;
  let gameRestored = false;
  let gameSkipped = false;
  if (backup.game) {
    if (isResumable(state.game)) {
      gameSkipped = true;
    } else {
      game = backup.game;
      gameRestored = true;
    }
  }

  return {
    state: { players, templates, game, history },
    summary: {
      playersAdded,
      playersUpdated,
      templatesAdded,
      templatesUpdated,
      gamesAdded,
      gamesUpdated,
      gameRestored,
      gameSkipped,
    },
  };
}
```

```ts
import { useEffect, useReducer, type Dispatch } from 'react';
import {
  addEntry as addGameEntry,
  removeEntry as removeGameEntry,
  undoLastEntry,
} from '../logic/game';
import { deserialize, serialize, type AppState } from '../logic/persistence';
import type { GameSession, HistoryEntry, Player, Template } from '../logic/types';

const STORAGE_KEY = 'boardgamescore.state.v1';

const EMPTY_STATE: AppState = { players: [], templates: [], game: null, history: [] };

export type AppAction =
  | { type: 'UPSERT_PLAYER'; player: Player }
  | { type: 'REMOVE_PLAYER'; id: string }
  | { type: 'UPSERT_TEMPLATE'; template: Template }
  | { type: 'REMOVE_TEMPLATE'; id: string }
  | { type: 'START_GAME'; game: GameSession }
  | { type: 'ADD_ENTRY'; playerId: string; categoryId: string; value: number; entryId: string }
  | { type: 'REMOVE_ENTRY'; playerId: string; categoryId: string; entryId: string }
  | { type: 'UNDO_LAST_ENTRY' }
  | { type: 'FINISH_GAME'; nowIso: string; historyEntry: HistoryEntry }
  | { type: 'IMPORT_MERGE'; state: AppState };

function upsertById<T extends { id: string }>(list: T[], item: T): T[] {
  const arr = list.slice();
  const i = arr.findIndex((x) => x.id === item.id);
  if (i >= 0) arr[i] = item;
  else arr.push(item);
  return arr;
}

export function appReducer(state: AppState, action: AppAction): AppState {
  switch (action.type) {
    case 'UPSERT_PLAYER':
      return { ...state, players: upsertById(state.players, action.player) };
    case 'REMOVE_PLAYER':
      return { ...state, players: state.players.filter((p) => p.id !== action.id) };
    case 'UPSERT_TEMPLATE':
      return { ...state, templates: upsertById(state.templates, action.template) };
    case 'REMOVE_TEMPLATE':
      return { ...state, templates: state.templates.filter((t) => t.id !== action.id) };
    case 'START_GAME':
      return { ...state, game: action.game };
    case 'ADD_ENTRY': {
      if (!state.game) return state;
      const game = addGameEntry(
        state.game,
        action.playerId,
        action.categoryId,
        action.value,
        action.entryId,
      );
      return { ...state, game };
    }
    case 'REMOVE_ENTRY': {
      if (!state.game) return state;
      const game = removeGameEntry(state.game, action.playerId, action.categoryId, action.entryId);
      return { ...state, game };
    }
    case 'UNDO_LAST_ENTRY': {
      if (!state.game) return state;
      return { ...state, game: undoLastEntry(state.game) };
    }
    case 'FINISH_GAME':
      return { ...state, game: null, history: [...state.history, action.historyEntry] };
    case 'IMPORT_MERGE':
      return action.state;
    default:
      return state;
  }
}

function loadInitialState(): AppState {
  return deserialize<AppState>(localStorage.getItem(STORAGE_KEY)) ?? EMPTY_STATE;
}

export function useAppState(): { state: AppState; dispatch: Dispatch<AppAction> } {
  const [state, dispatch] = useReducer(appReducer, undefined, loadInitialState);

  useEffect(() => {
    localStorage.setItem(STORAGE_KEY, serialize(state));
  }, [state]);

  return { state, dispatch };
}
```

```tsx
import { useState } from 'react';
import {
  Box,
  Button,
  IconButton,
  Stack,
  TextField,
  ToggleButton,
  ToggleButtonGroup,
  Typography,
} from '@mui/material';
import DeleteIcon from '@mui/icons-material/Delete';
import { evaluateFormula, parseFormula } from '../../logic/formula';
import { slugifyKey, validateTemplate } from '../../logic/template';
import type { Category, Template } from '../../logic/types';

interface DraftCategory {
  id: string;
  name: string;
  key: string;
  keyTouched: boolean;
}

function toDraftCategories(categories: Category[]): DraftCategory[] {
  return categories.map((c) => ({ ...c, keyTouched: true }));
}

export function TemplateBuilder({
  initialTemplate,
  onSave,
  onCancel,
}: {
  initialTemplate: Template | null;
  onSave: (template: Template) => void;
  onCancel: () => void;
}) {
  const [name, setName] = useState(initialTemplate?.name ?? '');
  const [categories, setCategories] = useState<DraftCategory[]>(
    toDraftCategories(initialTemplate?.categories ?? []),
  );
  const [formula, setFormula] = useState(initialTemplate?.formula ?? '');
  const [winDirection, setWinDirection] = useState<'high' | 'low'>(
    initialTemplate?.winDirection ?? 'high',
  );
  const [error, setError] = useState<string | null>(null);

  function addCategory() {
    setCategories((cs) => [
      ...cs,
      { id: crypto.randomUUID(), name: '', key: '', keyTouched: false },
    ]);
  }

  function removeCategory(id: string) {
    setCategories((cs) => cs.filter((c) => c.id !== id));
  }

  function updateCategoryName(id: string, newName: string) {
    setCategories((cs) =>
      cs.map((c) =>
        c.id === id ? { ...c, name: newName, key: c.keyTouched ? c.key : slugifyKey(newName) } : c,
      ),
    );
  }

  function updateCategoryKey(id: string, newKey: string) {
    setCategories((cs) =>
      cs.map((c) => (c.id === id ? { ...c, key: newKey, keyTouched: true } : c)),
    );
  }

  const knownKeys = categories.map((c) => c.key);
  const trimmedFormula = formula.trim();
  let formulaPreview: string;
  if (!trimmedFormula) {
    formulaPreview = 'No formula — total is the sum of all categories.';
  } else {
    const parsed = parseFormula(trimmedFormula, knownKeys);
    if (!parsed.ok) {
      formulaPreview = `Formula error: ${parsed.error}`;
    } else {
      const zeroValues = Object.fromEntries(knownKeys.map((k) => [k, 0]));
      formulaPreview = `Preview (with all categories at 0): ${evaluateFormula(parsed.ast, zeroValues)}`;
    }
  }

  function handleSave() {
    const finalCategories: Category[] = categories.map((c) => ({
      id: c.id,
      name: c.name.trim(),
      key: c.key,
    }));
    const result = validateTemplate({
      name,
      categories: finalCategories,
      formula: trimmedFormula || null,
    });
    if (!result.ok) {
      setError(result.error);
      return;
    }
    onSave({
      id: initialTemplate?.id ?? crypto.randomUUID(),
      name: name.trim(),
      categories: finalCategories,
      formula: trimmedFormula || null,
      winDirection,
      builtIn: false,
    });
  }

  return (
    <Box>
      <Typography variant="h5" gutterBottom>
        {initialTemplate ? 'Edit Template' : 'New Template'}
      </Typography>
      <Stack spacing={2} sx={{ maxWidth: 480 }}>
        <TextField label="Template Name" value={name} onChange={(e) => setName(e.target.value)} />

        <Typography variant="subtitle1">Categories</Typography>
        {categories.map((category) => (
          <Stack key={category.id} direction="row" spacing={1} alignItems="center">
            <TextField
              label="Category Name"
              value={category.name}
              onChange={(e) => updateCategoryName(category.id, e.target.value)}
            />
            <TextField
              label="Key"
              value={category.key}
              onChange={(e) => updateCategoryKey(category.id, e.target.value)}
              sx={{ width: 140 }}
            />
            <IconButton
              aria-label={`Remove category ${category.name || category.id}`}
              onClick={() => removeCategory(category.id)}
            >
              <DeleteIcon fontSize="small" />
            </IconButton>
          </Stack>
        ))}
        <Button onClick={addCategory}>Add Category</Button>

        <TextField
          label="Formula (optional)"
          value={formula}
          onChange={(e) => setFormula(e.target.value)}
          helperText={formulaPreview}
        />

        <Typography variant="subtitle1">Win Direction</Typography>
        <ToggleButtonGroup
          value={winDirection}
          exclusive
          onChange={(_, value: 'high' | 'low' | null) => value && setWinDirection(value)}
        >
          <ToggleButton value="high">Highest wins</ToggleButton>
```

```tsx
import type { Dispatch } from 'react';
import { Box, Button, Stack, Typography } from '@mui/material';
import { computeStandings } from '../../logic/template';
import { PlayerScoreCard } from './PlayerScoreCard';
import type { AppAction } from '../../hooks/useAppState';
import type { GameSession, HistoryEntry, Player, Template } from '../../logic/types';

export function GameScreen({
  game,
  template,
  players,
  dispatch,
  onFinish,
}: {
  game: GameSession;
  template: Template;
  players: Player[];
  dispatch: Dispatch<AppAction>;
  onFinish: (entry: HistoryEntry) => void;
}) {
  const standings = computeStandings(template, game, players);

  function handleFinish() {
    const nowIso = new Date().toISOString();
    const historyEntry: HistoryEntry = {
      gameId: game.id,
      templateName: template.name,
      playerNames: game.playerIds.map(
        (id) => players.find((p) => p.id === id)?.name ?? 'Unknown Player',
      ),
      finishedAt: nowIso,
      standings,
    };
    dispatch({ type: 'FINISH_GAME', nowIso, historyEntry });
    onFinish(historyEntry);
  }

  return (
    <Box>
      <Stack direction="row" justifyContent="space-between" alignItems="center" sx={{ mb: 2 }}>
        <Typography variant="h5">{template.name}</Typography>
        <Stack direction="row" spacing={1}>
          <Button
            onClick={() => dispatch({ type: 'UNDO_LAST_ENTRY' })}
            disabled={game.moveLog.length === 0}
          >
            Undo
          </Button>
          <Button variant="contained" onClick={handleFinish}>
            End Game
          </Button>
        </Stack>
      </Stack>
      {game.playerIds.map((playerId) => {
        const player = players.find((p) => p.id === playerId);
        if (!player) return null;
        const row = standings.find((s) => s.playerId === playerId)!;
        return (
          <PlayerScoreCard
            key={playerId}
            player={player}
            categories={template.categories}
            entriesByCategory={game.entries[playerId] ?? {}}
            total={row.total}
            rank={row.rank}
            onAddEntry={(categoryId, value) =>
              dispatch({
                type: 'ADD_ENTRY',
                playerId,
                categoryId,
                value,
                entryId: crypto.randomUUID(),
              })
            }
            onRemoveEntry={(categoryId, entryId) =>
              dispatch({ type: 'REMOVE_ENTRY', playerId, categoryId, entryId })
            }
          />
        );
      })}
    </Box>
  );
}
```

```tsx
import { useRef, useState, type Dispatch } from 'react';
import { Alert, Box, Button, Stack, Typography } from '@mui/material';
import { buildBackup, mergeBackup, validateBackup, type AppState } from '../../logic/persistence';
import type { AppAction } from '../../hooks/useAppState';

export function SettingsScreen({
  state,
  dispatch,
}: {
  state: AppState;
  dispatch: Dispatch<AppAction>;
}) {
  const [message, setMessage] = useState<{ severity: 'success' | 'error'; text: string } | null>(
    null,
  );
  const fileInputRef = useRef<HTMLInputElement>(null);

  function handleExport() {
    const backup = buildBackup(state, Date.now());
    const blob = new Blob([JSON.stringify(backup, null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = `boardgamescore-backup-${backup.exportedAt.slice(0, 10)}.json`;
    link.click();
    URL.revokeObjectURL(url);
  }

  async function handleImportFile(file: File) {
    let parsed: unknown;
    try {
      parsed = JSON.parse(await file.text());
    } catch {
      setMessage({ severity: 'error', text: 'That file is not valid JSON.' });
      return;
    }
    const validated = validateBackup(parsed);
    if (!validated.ok) {
      setMessage({ severity: 'error', text: validated.reason });
      return;
    }
    const { state: merged, summary } = mergeBackup(state, validated.backup);
    dispatch({ type: 'IMPORT_MERGE', state: merged });
    setMessage({
      severity: 'success',
      text: `Imported ${summary.playersAdded} new / ${summary.playersUpdated} updated players, ${summary.templatesAdded} new / ${summary.templatesUpdated} updated templates, ${summary.gamesAdded} new / ${summary.gamesUpdated} updated games.${summary.gameSkipped ? ' In-progress game skipped (one is already active).' : ''}`,
    });
  }

  return (
    <Box>
      <Typography variant="h5" gutterBottom>
        Settings
      </Typography>
      <Stack direction="row" spacing={1} sx={{ mb: 2 }}>
        <Button variant="contained" onClick={handleExport}>
          Export Backup
        </Button>
        <Button onClick={() => fileInputRef.current?.click()}>Import Backup</Button>
        <input
          ref={fileInputRef}
          type="file"
          accept="application/json"
          hidden
          aria-label="Import Backup File"
          onChange={(e) => {
            const file = e.target.files?.[0];
            if (file) void handleImportFile(file);
            e.target.value = '';
          }}
        />
      </Stack>
      {message && <Alert severity={message.severity}>{message.text}</Alert>}
    </Box>
  );
}
```

```tsx
import { useState } from 'react';
import { Box, CssBaseline, ThemeProvider } from '@mui/material';
import { createAppTheme } from '../theme';
import { useAppState } from '../hooks/useAppState';
import { getAllTemplates } from '../logic/template';
import type { HistoryEntry, Template } from '../logic/types';
import { Nav, type Screen } from './Nav';
import { HomeScreen } from './Home/HomeScreen';
import { PlayersScreen } from './Players/PlayersScreen';
import { TemplatesScreen } from './Templates/TemplatesScreen';
import { TemplateBuilder } from './Templates/TemplateBuilder';
import { GameScreen } from './Game/GameScreen';
import { SummaryScreen } from './Summary/SummaryScreen';
import { HistoryScreen } from './History/HistoryScreen';
import { SettingsScreen } from './Settings/SettingsScreen';

export default function App() {
  const { state, dispatch } = useAppState();
  const [screen, setScreen] = useState<Screen>('home');
  const [editingTemplate, setEditingTemplate] = useState<Template | null>(null);
  const [viewingHistoryEntry, setViewingHistoryEntry] = useState<HistoryEntry | null>(null);

  const allTemplates = getAllTemplates(state.templates);
  const activeTemplate = state.game
    ? allTemplates.find((t) => t.id === state.game!.templateId)
    : undefined;

  return (
    <ThemeProvider theme={createAppTheme('light')}>
      <CssBaseline />
      <Box sx={{ maxWidth: 720, mx: 'auto', p: 2 }}>
        <Nav screen={screen} onNavigate={setScreen} />

        {screen === 'home' && (
          <HomeScreen
            templates={allTemplates}
            players={state.players}
            game={state.game}
            dispatch={dispatch}
            onStartGame={(game) => {
              dispatch({ type: 'START_GAME', game });
              setScreen('game');
            }}
            onResume={() => setScreen('game')}
          />
        )}

        {screen === 'players' && <PlayersScreen players={state.players} dispatch={dispatch} />}

        {screen === 'templates' && (
          <TemplatesScreen
            templates={allTemplates}
            dispatch={dispatch}
            onNew={() => {
              setEditingTemplate(null);
              setScreen('templateBuilder');
            }}
            onEdit={(template) => {
              setEditingTemplate(template);
              setScreen('templateBuilder');
            }}
            onDuplicate={(template) => {
              setEditingTemplate({
                ...template,
                id: crypto.randomUUID(),
                name: `${template.name} (copy)`,
                builtIn: false,
              });
              setScreen('templateBuilder');
            }}
          />
        )}

        {screen === 'templateBuilder' && (
          <TemplateBuilder
            initialTemplate={editingTemplate}
            onSave={(template) => {
              dispatch({ type: 'UPSERT_TEMPLATE', template });
              setScreen('templates');
            }}
            onCancel={() => setScreen('templates')}
          />
        )}

        {screen === 'game' && state.game && activeTemplate && (
          <GameScreen
            game={state.game}
            template={activeTemplate}
            players={state.players}
            dispatch={dispatch}
            onFinish={(entry) => {
              setViewingHistoryEntry(entry);
              setScreen('summary');
            }}
          />
        )}

        {screen === 'summary' && viewingHistoryEntry && (
          <SummaryScreen entry={viewingHistoryEntry} onDone={() => setScreen('home')} />
        )}

        {screen === 'history' && (
          <HistoryScreen
            history={state.history}
            onSelect={(entry) => {
              setViewingHistoryEntry(entry);
              setScreen('summary');
            }}
          />
        )}

        {screen === 'settings' && <SettingsScreen state={state} dispatch={dispatch} />}
      </Box>
    </ThemeProvider>
  );
}
```
